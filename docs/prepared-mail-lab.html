HOST INTEGRATION

Prepare once. Check. Submit.

The host owns routing, policy, authenticated approval, and the mail service. The sender callback must submit the prepared request without adding a destination afterward.

import { createPreparedMailToolExecute } from "./pilots/action-boundary/prepared-mail-tool.mjs";
import { mailTargetResource } from "./pilots/action-boundary/mail-target-manifest.mjs";

const intended = mailTargetResource({ to: "team@example.test" });
const policy = { protocol: "aml-action-policy/1", rules: [{
  tool: "send_message", effect: "send", resource: intended.resource,
  requires_approval: true
}] };
const execute = createPreparedMailToolExecute({
  purpose: "Send release", policy,
  prepare: (toolArgs, context) => hostBuildProviderRequest(toolArgs, context),
  requestApproval: hostAuthenticatedApproval,
  consumeApproval: hostAtomicOneUseClaim,
  execute: preparedRequest => hostMailer.send(preparedRequest)
});
